// Inserted before the preserved host/core dispatcher; no game assets are rewritten.
const accountEnv=env=>({...env,ACCOUNTS_MODE:'email',STRIPE_IDENTITY_POLICY_CONFIRMED:'0',AGE_ELIGIBILITY_POLICY_CONFIRMED:'0'});
const accountPaths=new Set(['/account','/account/','/account/index.html']);
function privateAccountResponse(response,gated=false){const headers=new Headers(response.headers);headers.set('Content-Security-Policy',ACCOUNT_CSP);headers.set('Referrer-Policy','no-referrer');headers.set('Cache-Control','private, no-store');headers.set('X-Content-Type-Options','nosniff');headers.set('X-Robots-Tag','noindex');headers.append('Vary','Cookie');let result=new Response(response.body,{status:response.status,statusText:response.statusText,headers});if(gated)result=new HTMLRewriter().on('html',{element(e){e.setAttribute('data-account-gate','1')}}).transform(result);return result;}
async function accountPage(request,env,gated=false){const u=new URL(request.url);u.pathname='/account/index.html';u.search='';const headers=new Headers(request.headers);for(const name of ['Range','If-Range','If-None-Match','If-Modified-Since','If-Match','If-Unmodified-Since'])headers.delete(name);const response=privateAccountResponse(await env.ASSETS.fetch(new Request(u,{method:request.method,headers})),gated);return request.method==='HEAD'?new Response(null,{status:response.status,statusText:response.statusText,headers:response.headers}):response;}
async function accountHtml(request,env){const url=new URL(request.url);if(url.hostname==='banderov-backend.alexanderbanderov.workers.dev'&&['GET','HEAD'].includes(request.method)&&ACCOUNT_HTML_PATHS.has(url.pathname)&&!['/','/index.html'].includes(url.pathname)&&!accountPaths.has(url.pathname))return env.ASSETS.fetch(request);if(url.hostname!=='banderov.com'||!['GET','HEAD'].includes(request.method))return null;if(accountPaths.has(url.pathname))return accountPage(request,env);if(!ACCOUNT_HTML_PATHS.has(url.pathname))return null;let blocked=false;const cookie=request.headers.get('Cookie')||'';if(/(?:^|;\s*)__Host-banderov-session=/.test(cookie)){try{const current=new URL('/api/account/current',url);const status=await emailAccounts(new Request(current,{headers:{Cookie:cookie,Accept:'application/json'}}),accountEnv(env));const data=await status.json();blocked=!(status.ok&&data.version===3&&data.authenticated===false)&&!(status.ok&&data.version===3&&data.authenticated===true&&data.account?.status==='active'&&data.account.identityVerified===true&&data.account.ageEligibility==='eligible_18plus');}catch{blocked=true;}}if(blocked)return accountPage(request,env,true);const assetUrl=new URL(request.url);if(assetUrl.pathname==='/index.html')assetUrl.pathname='/';let response=await env.ASSETS.fetch(new Request(assetUrl,request));if(response.headers.get('Content-Type')?.includes('text/html')){const headers=new Headers(response.headers);headers.set('Cache-Control','private, no-store');headers.append('Vary','Cookie');response=new Response(response.body,{status:response.status,statusText:response.statusText,headers});if(request.headers.get('Sec-Fetch-Dest')==='document'||request.headers.get('Accept')?.includes('text/html'))response=new HTMLRewriter().on('head',{element(e){e.append('<link rel="stylesheet" href="/account/site-gate.css"><script src="/account/site-gate.js" defer></script>',{html:true})}}).transform(response);}return response;}
